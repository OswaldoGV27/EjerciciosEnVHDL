library ieee;
    use ieee.std_logic_1164.all;

    entity BIN_GRAY is port (A, B, C, D: in std_logic;
        G0: out std_logic;
        G1: out std_logic;
        G2: out std_logic;
        G3: out std_logic
    );
    end BIN_GRAY;

    architecture ABIN_GRAY of BIN_GRAY is begin
        G0 <= ((not C) and D) or (C and (not D));
        G1 <= ((not B) and C) or (B and (not C));
        G2 <= ((not A) and B) or (A and (not B));
        G3 <= (A);
    end ABIN_GRAY;
